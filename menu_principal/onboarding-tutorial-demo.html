// DESBLOQUEAR SCROLL IMEDIATAMENTE E PERIODICAMENTE function unlockScroll() { document.body.style.overflow = ''; document.body.style.position = ''; document.body.style.top = ''; document.body.style.width = ''; document.documentElement.style.overflow = ''; document.documentElement.style.position = ''; // Remover overlays que podem estar bloqueando const overlays = document.querySelectorAll('.onboarding-overlay, #onboarding-overlay'); overlays.forEach(ov => { ov.style.pointerEvents = 'none'; ov.style.display = 'none'; }); // Remover event listeners de scroll bloqueado const preventers = window._scrollPreventers; if (preventers) { window.removeEventListener('wheel', preventers.wheel, { capture: true }); window.removeEventListener('touchmove', preventers.touchmove, { capture: true }); window.removeEventListener('keydown', preventers.keydown, { capture: true }); window._scrollPreventers = null; } } // Desbloquear imediatamente unlockScroll(); // Desbloquear periodicamente - MAIS FREQUENTE setInterval(unlockScroll, 100); // Desbloquear quando a página carregar if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', unlockScroll); } else { unlockScroll(); } window.addEventListener('load', unlockScroll); // FORÇAR DESBLOQUEIO A CADA EVENTO DE SCROLL window.addEventListener('wheel', function(e) { unlockScroll(); }, { passive: true }); window.addEventListener('touchmove', function(e) { unlockScroll(); }, { passive: true }); // FORÇAR DESBLOQUEIO NO KEYDOWN window.addEventListener('keydown', function(e) { if ([32, 33, 34, 35, 36, 37, 38, 39, 40].includes(e.keyCode)) { unlockScroll(); } }, { passive: true });
